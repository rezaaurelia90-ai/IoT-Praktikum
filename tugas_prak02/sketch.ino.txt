#include <DHT.h>

// Definisi Pin Komponen pada ESP32
const byte ldrPin = 34;   // Pin GPIO 34 (Analog Input LDR)
const byte dhtPin = 13;   // Pin GPIO 13 (Data Sensor DHT11)
const byte relayPin = 12; // Pin GPIO 12 (Sinyal Modul Relay)
const byte ledPin = 5;    // Pin GPIO 5 (LED Indikator)

#define DHTTYPE DHT11
DHT dht(dhtPin, DHTTYPE);

void setup() {
  Serial.begin(115200);
  dht.begin();

  // Konfigurasi Pin Output
  pinMode(relayPin, OUTPUT);
  pinMode(ledPin, OUTPUT);

  // Status Keamanan Awal: Aktuator MATI
  // Catatan: Pada Relay Active-Low, HIGH berarti MATI
  digitalWrite(relayPin, HIGH); 
  digitalWrite(ledPin, LOW);

  Serial.println("--- Sistem Otomasi Smart Warehouse (ESP32) Dimulai ---");
}

void loop() {
  delay(2000); // Jeda pembacaan DHT11 (2 detik)

  // Membaca data sensor
  int ldrValue = analogRead(ldrPin);
  float temp = dht.readTemperature();
  float hum = dht.readHumidity();

  // Penanganan Error jika sensor DHT11 tidak terhubung
  if (isnan(temp) || isnan(hum)) {
    Serial.println("ERROR: Gagal membaca data dari sensor DHT!");
    return;
  }

  // Tampilkan telemetri data ke Serial Monitor
  Serial.print("ADC LDR: ");
  Serial.print(ldrValue);
  Serial.print(" | Suhu: ");
  Serial.print(temp);
  Serial.print(" C | Kelembapan: ");
  Serial.print(hum);
  Serial.println(" %");

  // Rule Engine: Logika OR (||)
  // Aktuator aktif jika Suhu > 34 C ATAU Cahaya LDR < 300
  if (temp > 34.0 || ldrValue < 300) {
    digitalWrite(relayPin, LOW);  // Relay Aktif (Active-Low)
    digitalWrite(ledPin, HIGH);   // LED Indikator Menyala
    Serial.println("PERINGATAN: Kondisi Buruk! Aktuator Relay & LED AKTIF!");
  } else {
    digitalWrite(relayPin, HIGH); // Relay Mati (Active-Low)
    digitalWrite(ledPin, LOW);    // LED Indikator Padam
    Serial.println("STATUS: Lingkungan Aman.");
  }

  Serial.println("----------------------------------------------------");
}
